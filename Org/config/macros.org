# set up acronyms
#+MACRO: gls @@latex:\gls{$1}@@
#+MACRO: glspl @@latex:\glspl{$1}@@
#+MACRO: Gls @@latex:\Gls{$1}@@

# declare acronyms macro
#+MACRO: setac @@latex:\newacronym{$1}{$2}{$3}@@

# image macro
#+MACRO: figure @@latex:\begin{figure}[h] \centering \includegraphics[width=0.95\columnwidth]{Figures/$1} \caption{$2} \end{figure}@@
#+MACRO: figurelabel @@latex:\begin{figure}[h] \centering \includegraphics[width=0.95\columnwidth]{Figures/$1} \caption{$2} \label{$3} \end{figure}@@

#+MACRO: figure-group @@latex:\begin{figure}[h] \centering \begin{subfigure}[b]{0.48\columnwidth} \centering \includegraphics[width=\textwidth]{Figures/$1} \caption{$2} \end{subfigure} \hfill \begin{subfigure}[b]{0.48\columnwidth} \centering \includegraphics[width=\textwidth]{Figures/$3} \caption{$4} \end{subfigure} \caption{$5} \end{figure}@@

# citations
#+MACRO: cite @@latex:\cite{$1}@@
#+MACRO: citep @@latex:\citep{$1}@@
#+MACRO: citet @@latex:\citet{$1}@@
#+MACRO: citewp @@latex:\cite[$1]{$2}@@

#+MACRO: mark @@latex:\texttt{$1}@@
#+MACRO: latex @@latex:$1@@

#+MACRO: ref @@latex:\ref{$1}@@
